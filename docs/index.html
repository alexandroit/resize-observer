@revivejs/resize-observer

A framework-agnostic ResizeObserver ponyfill with content-box, border-box, and device-pixel-content-box support.

Repository Issues npm

Install

npm install @revivejs/resize-observer

Quick start

import { ResizeObserver } from '@revivejs/resize-observer';

const ro = new ResizeObserver((entries) => {
  for (const entry of entries) {
    const { inlineSize, blockSize } = entry.contentBoxSize[0];
    entry.target.textContent = `${Math.round(inlineSize)} x ${Math.round(blockSize)}`;
  }
});

ro.observe(document.querySelector('[data-resize]'));
